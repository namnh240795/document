CREATE TABLE xxx_users (
  id            BIGSERIAL PRIMARY KEY,
  username      VARCHAR(50) NOT NULL UNIQUE,
  email         VARCHAR(100) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  full_name     VARCHAR(100),
  phone         VARCHAR(20),

  created_at    TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at    TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE xxx_orders (
  id                    BIGSERIAL PRIMARY KEY,
  user_id               BIGINT NOT NULL REFERENCES xxx_users(id),
  order_number          VARCHAR(20) NOT NULL UNIQUE,
  status                VARCHAR(20) NOT NULL DEFAULT 'pending'
                        CHECK (status IN ('pending','confirmed','processing','shipped','delivered','cancelled')),
  subtotal              DECIMAL(10,2) NOT NULL DEFAULT 0,
  tax_amount            DECIMAL(10,2) NOT NULL DEFAULT 0,
  shipping_fee          DECIMAL(10,2) NOT NULL DEFAULT 0,
  discount_amount       DECIMAL(10,2) NOT NULL DEFAULT 0,
  total_amount          DECIMAL(10,2) NOT NULL DEFAULT 0,
  shipping_address_id   BIGINT REFERENCES xxx_addresses(id),
  notes                 TEXT,
  created_at            TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at            TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: xxx_order_items

CREATE TABLE xxx_order_items (

  order_id    BIGINT NOT NULL REFERENCES xxx_orders(id) ON DELETE CASCADE,

2.5 Database: ecom_db - Product Catalog Module

Modules: PROD | Resolves: FR-003

Table: xxx_products

CREATE TABLE xxx_products (
  id              BIGSERIAL PRIMARY KEY,

2.7 Database: wallet_db - Wallet Module

Modules: WAL | Resolves: FR-008, FR-009, FR-010

Table: xxx_wallets

CREATE TABLE xxx_wallets (
  id          BIGSERIAL PRIMARY KEY,
  user_id     BIGINT NOT NULL REFERENCES xxx_users(id) UNIQUE,
  balance     DECIMAL(12,2) NOT NULL DEFAULT 0 CHECK (balance >= 0),

              CHECK (status IN ('active','frozen','closed')),

Table: xxx_wallet_transactions

CREATE TABLE xxx_wallet_transactions (
  id              BIGSERIAL PRIMARY KEY,
  wallet_id       BIGINT NOT NULL REFERENCES xxx_wallets(id),
  type            VARCHAR(20) NOT NULL
                  CHECK (type IN ('topup','payment','refund','transfer_in','transfer_out','withdrawal')),
  amount          DECIMAL(12,2) NOT NULL,
  balance_after   DECIMAL(12,2) NOT NULL,
  reference_type  VARCHAR(50),
  reference_id    BIGINT,
  description     TEXT,
  created_at      TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: xxx_wallet_topup_requests

CREATE TABLE xxx_wallet_topup_requests (
  id            BIGSERIAL PRIMARY KEY,
  wallet_id     BIGINT NOT NULL REFERENCES xxx_wallets(id),
  amount        DECIMAL(12,2) NOT NULL CHECK (amount > 0),
  method        VARCHAR(20) NOT NULL
                CHECK (method IN ('bank_transfer','credit_card','e_wallet')),
  status        VARCHAR(20) NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending','completed','failed')),
  transaction_id VARCHAR(100),
  created_at    TIMESTAMP NOT NULL DEFAULT NOW(),
  completed_at  TIMESTAMP
);

3. Indexes

3.1 Index Strategy

  • Primary keys: auto-created
  • Foreign keys: always index
  • Frequently queried columns: index
  • UNIQUE constraints: auto-create index

3.2 Index Definitions

-- MODULE: Authentication
CREATE INDEX idx_xxx_users_email ON xxx_users(email);
CREATE INDEX idx_xxx_users_role ON xxx_users(role);
CREATE INDEX idx_xxx_users_status ON xxx_users(status);

-- MODULE: E-Commerce
CREATE INDEX idx_xxx_orders_user_id ON xxx_orders(user_id);
CREATE INDEX idx_xxx_orders_status ON xxx_orders(status);
CREATE INDEX idx_xxx_orders_created_at ON xxx_orders(created_at DESC);
CREATE INDEX idx_xxx_orders_user_status ON xxx_orders(user_id, status);
CREATE INDEX idx_xxx_order_items_order_id ON xxx_order_items(order_id);
CREATE INDEX idx_xxx_order_items_product_id ON xxx_order_items(product_id);

-- MODULE: Product Catalog
CREATE INDEX idx_xxx_products_category_id ON xxx_products(category_id);
CREATE INDEX idx_xxx_products_status ON xxx_products(status);
CREATE INDEX idx_xxx_products_price ON xxx_products(price);
CREATE INDEX idx_xxx_products_sku ON xxx_products(sku);
CREATE INDEX idx_xxx_categories_parent_id ON xxx_categories(parent_id);
CREATE INDEX idx_xxx_categories_slug ON xxx_categories(slug);

-- MODULE: Payment
CREATE INDEX idx_xxx_payments_order_id ON xxx_payments(order_id);
CREATE INDEX idx_xxx_payments_status ON xxx_payments(status);

-- MODULE: Wallet
CREATE INDEX idx_xxx_wallets_user_id ON xxx_wallets(user_id);
CREATE INDEX idx_xxx_wallet_tx_wallet_id ON xxx_wallet_transactions(wallet_id);
CREATE INDEX idx_xxx_wallet_tx_type ON xxx_wallet_transactions(type);
CREATE INDEX idx_xxx_wallet_tx_created ON xxx_wallet_transactions(created_at DESC);
CREATE INDEX idx_xxx_topup_wallet_id ON xxx_wallet_topup_requests(wallet_id);
CREATE INDEX idx_xxx_topup_status ON xxx_wallet_topup_requests(status);

7. Performance Tuning

7.1 PostgreSQL Configuration

# postgresql.conf
shared_buffers = 4GB            # 25% of RAM
effective_cache_size = 12GB     # 75% of RAM
work_mem = 64MB
maintenance_work_mem = 1GB
max_connections = 200
random_page_cost = 1.1          # SSD
effective_io_concurrency = 200  # SSD

7.2 Query Optimization Notes

  • Use EXPLAIN ANALYZE for slow queries
  • Avoid N+1 queries (use JOIN or batch loading)
  • Use connection pooling (PgBouncer)
  • Monitor slow query log (> 100ms)
