2.2 Table: sms_logs

ColumnTypeNullableDefaultDescription
iduuidNOgen_random_uuid()Primary key
user_iduuidYESNULLAssociated user (from AUTH)
phonevarchar(20)NO-Recipient phone number
messagetextNO-SMS message content
otp_codevarchar(10)YESNULLOTP code (if applicable)
statusvarchar(20)NOpendingpending, sent, delivered, failed
providervarchar(50)NOtwilioSMS provider name
provider_message_idvarchar(100)YESNULLProvider's message ID
error_messagetextYESNULLError message if failed
retry_countintegerNO0Number of retry attempts
expires_attimestampYESNULLOTP expiry time
sent_attimestampYESNULLTime SMS was sent
delivered_attimestampYESNULLTime SMS was delivered
created_attimestampNONOW()Record creation time
CREATE TABLE sms_logs (
  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id               UUID,
  phone                 VARCHAR(20) NOT NULL,
  message               TEXT NOT NULL,
  otp_code              VARCHAR(10),
  status                VARCHAR(20) NOT NULL DEFAULT 'pending'
                        CHECK (status IN ('pending', 'sent', 'delivered', 'failed')),
  provider              VARCHAR(50) NOT NULL DEFAULT 'twilio',
  provider_message_id   VARCHAR(100),
  error_message         TEXT,
  retry_count           INTEGER NOT NULL DEFAULT 0,
  expires_at            TIMESTAMP,
  sent_at               TIMESTAMP,
  delivered_at          TIMESTAMP,
  created_at            TIMESTAMP NOT NULL DEFAULT NOW()
);
