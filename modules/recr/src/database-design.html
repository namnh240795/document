1. Overview

1.1 Purpose

This document defines the database design for the Recruitment module (database: recr_db).

2.2 Tables

Table: recr_users - Core User Table (All Roles)

Note: User authentication (email, password, role) is handled by AUTH module. This table only stores recruitment-specific data and links to auth_users via user_id.

ColumnTypeNullableDefaultDescription
idbigintNOnextval('recr_users_id_seq')Primary key
user_idbigintNO-FK to auth_users.id (UNIQUE)
avatar_urlvarchar(500)YES-Avatar image URL
created_attimestampNONOW()Creation timestamp
updated_attimestampNONOW()Last update timestamp
CREATE TABLE recr_users (
  id          BIGSERIAL PRIMARY KEY,
  user_id     BIGINT NOT NULL UNIQUE REFERENCES auth_users(id),
  avatar_url  VARCHAR(500),
  created_at  TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at  TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_candidates - Candidate Profile

Note: Resume storage has been migrated to the FILE module. The resume_url column is deprecated. Resume file references are now stored in the recr_candidate_resumes table via file_id (UUID referencing sys_files.id in the FILE module).

CREATE TABLE recr_candidates (
  id               BIGSERIAL PRIMARY KEY,
  user_id          BIGINT NOT NULL UNIQUE REFERENCES recr_users(id),
  resume_url       VARCHAR(500),              -- DEPRECATED: use recr_candidate_resumes instead
  linkedin_url     VARCHAR(300),
  skills           JSONB DEFAULT '[]',
  experience_years INT DEFAULT 0,
  current_company  VARCHAR(200),
  expected_salary  DECIMAL(12,2),
  source           VARCHAR(20) CHECK (source IN ('website','referral','linkedin','job_board','other')),
  created_at       TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at       TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_candidate_resumes - Candidate Resume List (FILE Module Integration)

Description: Stores references to candidate resume files managed by the FILE module. Each candidate can have multiple resumes. The file_id references sys_files.id in the FILE module's database.

CREATE TABLE recr_candidate_resumes (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  candidate_id    BIGINT NOT NULL REFERENCES recr_candidates(id) ON DELETE CASCADE,
  file_id         UUID NOT NULL REFERENCES sys_files(id) ON DELETE CASCADE,
  is_primary      BOOLEAN NOT NULL DEFAULT FALSE,
  created_at      TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at      TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Indexes for resume management
CREATE INDEX idx_recr_candidate_resumes_candidate_id ON recr_candidate_resumes(candidate_id);
CREATE INDEX idx_recr_candidate_resumes_file_id ON recr_candidate_resumes(file_id);
CREATE INDEX idx_recr_candidate_resumes_primary ON recr_candidate_resumes(candidate_id, is_primary) WHERE is_primary = TRUE;

Table: recr_employers - Employer Profile

CREATE TABLE recr_employers (
  id                  BIGSERIAL PRIMARY KEY,
  user_id             BIGINT NOT NULL UNIQUE REFERENCES recr_users(id),
  company_name        VARCHAR(200) NOT NULL,
  company_website     VARCHAR(300),
  company_description TEXT,
  industry            VARCHAR(100),
  company_size        VARCHAR(20) CHECK (company_size IN ('1-10','11-50','51-200','201-500','501-1000','1000+')),
  logo_url            VARCHAR(500),
  is_verified         BOOLEAN DEFAULT false,
  created_at          TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at          TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_hr_profiles - HR Profile

CREATE TABLE recr_hr_profiles (
  id          BIGSERIAL PRIMARY KEY,
  user_id     BIGINT NOT NULL UNIQUE REFERENCES recr_users(id),
  department  VARCHAR(100),
  position    VARCHAR(100),
  created_at  TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at  TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_jobs - Job Listings

Note: Both HR and Employer can create jobs. The created_by field tracks who created the job (HR or Employer).

CREATE TABLE recr_jobs (
  id              BIGSERIAL PRIMARY KEY,
  employer_id     BIGINT REFERENCES recr_employers(id),  -- NULL for HR-created jobs
  created_by      BIGINT NOT NULL REFERENCES recr_users(id),  -- HR or Employer user_id
  title           VARCHAR(200) NOT NULL,
  description     TEXT,
  department      VARCHAR(100),
  location        VARCHAR(200),
  salary_min      DECIMAL(12,2),
  salary_max      DECIMAL(12,2),
  employment_type VARCHAR(20) NOT NULL CHECK (employment_type IN ('full_time','part_time','contract','internship')),
  status          VARCHAR(20) NOT NULL DEFAULT 'draft' CHECK (status IN ('draft','pending_approval','open','closed','archived')),
  approved_by     BIGINT REFERENCES recr_users(id),
  approved_at     TIMESTAMP,
  created_at      TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at      TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_applications - Job Applications

Note: The resume_file_id column stores a frozen (immutable) copy of the resume file reference at submission time. This ensures the resume reviewed by HR/Employer is the same one the candidate submitted, even if the candidate later updates or deletes the original resume from their resume list.

CREATE TABLE recr_applications (
  id              BIGSERIAL PRIMARY KEY,
  job_id          BIGINT NOT NULL REFERENCES recr_jobs(id),
  candidate_id    BIGINT NOT NULL REFERENCES recr_candidates(id),
  resume_file_id  UUID REFERENCES sys_files(id),  -- Frozen resume at submission (FILE module)
  cover_letter    TEXT,
  status          VARCHAR(20) NOT NULL DEFAULT 'submitted' CHECK (status IN ('submitted','screening','interview','offer','rejected','hired')),
  applied_at      TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at      TIMESTAMP NOT NULL DEFAULT NOW(),
  UNIQUE(job_id, candidate_id)
);

Table: recr_interviews - Interview Schedule

CREATE TABLE recr_interviews (
  id               BIGSERIAL PRIMARY KEY,
  application_id   BIGINT NOT NULL REFERENCES recr_applications(id),
  interviewer_id   BIGINT NOT NULL REFERENCES recr_users(id),
  scheduled_at     TIMESTAMP NOT NULL,
  duration_minutes INT DEFAULT 60,
  location         VARCHAR(200),
  meeting_link     VARCHAR(500),
  status           VARCHAR(20) NOT NULL DEFAULT 'scheduled' CHECK (status IN ('scheduled','completed','cancelled','no_show')),
  notes            TEXT,
  created_at       TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_evaluations - Interview Evaluations

CREATE TABLE recr_evaluations (
  id                     BIGSERIAL PRIMARY KEY,
  interview_id           BIGINT NOT NULL REFERENCES recr_interviews(id),
  evaluator_id           BIGINT NOT NULL REFERENCES recr_users(id),
  score                  INT CHECK (score >= 1 AND score <= 10),
  technical_score        INT CHECK (technical_score >= 1 AND technical_score <= 10),
  communication_score    INT CHECK (communication_score >= 1 AND communication_score <= 10),
  cultural_fit_score     INT CHECK (cultural_fit_score >= 1 AND cultural_fit_score <= 10),
  recommendation         VARCHAR(20) CHECK (recommendation IN ('strong_yes','yes','neutral','no','strong_no')),
  comments               TEXT,
  submitted_at           TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_job_requirements - Job Requirements

CREATE TABLE recr_job_requirements (
  id                BIGSERIAL PRIMARY KEY,
  job_id            BIGINT NOT NULL REFERENCES recr_jobs(id),
  requirement_type  VARCHAR(20) NOT NULL CHECK (requirement_type IN ('skill','experience','education','certification')),
  description       TEXT NOT NULL,
  is_mandatory      BOOLEAN DEFAULT true,
  sort_order        INT DEFAULT 0
);

Table: recr_subscription_plans - Subscription Plans

Description: Defines available subscription plans with pricing and discounts.

CREATE TABLE recr_subscription_plans (
  id                BIGSERIAL PRIMARY KEY,
  name              VARCHAR(50) NOT NULL UNIQUE,           -- '1_month', '6_months', '1_year'
  display_name      VARCHAR(100) NOT NULL,                 -- '1 Month', '6 Months', '1 Year'
  duration_months   INT NOT NULL,                          -- 1, 6, 12
  base_price        DECIMAL(12,2) NOT NULL,                -- Price before discount
  discount_percent  INT DEFAULT 0 CHECK (discount_percent >= 0 AND discount_percent <= 100),
  free_months       INT DEFAULT 0,                         -- 0 for 1mo/6mo, 2 for 1yr
  final_price       DECIMAL(12,2) NOT NULL,                -- Price after discount
  is_active         BOOLEAN DEFAULT true,
  features          JSONB DEFAULT '[]',                    -- List of features included
  created_at        TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_subscriptions - User Subscriptions

Description: Tracks user subscriptions and access control.

CREATE TABLE recr_subscriptions (
  id                BIGSERIAL PRIMARY KEY,
  user_id           BIGINT NOT NULL REFERENCES recr_users(id),
  plan_id           BIGINT NOT NULL REFERENCES recr_subscription_plans(id),
  status            VARCHAR(20) NOT NULL DEFAULT 'active' CHECK (status IN ('active','expired','cancelled','pending')),
  start_date        TIMESTAMP NOT NULL,
  end_date          TIMESTAMP NOT NULL,
  payment_amount    DECIMAL(12,2) NOT NULL,                -- Actual amount paid
  payment_method    VARCHAR(50),                           -- 'credit_card', 'bank_transfer', 'e_wallet'
  payment_reference VARCHAR(100),                          -- Transaction ID
  auto_renew        BOOLEAN DEFAULT false,
  created_at        TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMP NOT NULL DEFAULT NOW(),
  CHECK (end_date > start_date)
);

-- Index for subscription validation
CREATE INDEX idx_recr_subscriptions_user_id ON recr_subscriptions(user_id);
CREATE INDEX idx_recr_subscriptions_status ON recr_subscriptions(status);
CREATE UNIQUE INDEX idx_recr_subscriptions_active ON recr_subscriptions(user_id) WHERE status = 'active';

Table: recr_subscription_history - Subscription History

Description: Logs all subscription changes for audit trail.

CREATE TABLE recr_subscription_history (
  id                BIGSERIAL PRIMARY KEY,
  subscription_id   BIGINT NOT NULL REFERENCES recr_subscriptions(id),
  action            VARCHAR(50) NOT NULL CHECK (action IN ('created','renewed','upgraded','cancelled','expired','payment_received')),
  old_status        VARCHAR(20),
  new_status        VARCHAR(20),
  old_plan_id       BIGINT REFERENCES recr_subscription_plans(id),
  new_plan_id       BIGINT REFERENCES recr_subscription_plans(id),
  notes             TEXT,
  created_at        TIMESTAMP NOT NULL DEFAULT NOW()
);

Table: recr_job_reviews - Job Reviews

Description: Candidate reviews and ratings for jobs they applied to.

CREATE TABLE recr_job_reviews (
  id                BIGSERIAL PRIMARY KEY,
  job_id            BIGINT NOT NULL REFERENCES recr_jobs(id),
  user_id           BIGINT NOT NULL REFERENCES recr_users(id),
  overall_rating    INT NOT NULL CHECK (overall_rating >= 1 AND overall_rating <= 5),
  work_life_balance INT CHECK (work_life_balance >= 1 AND work_life_balance <= 5),
  compensation      INT CHECK (compensation >= 1 AND compensation <= 5),
  career_growth     INT CHECK (career_growth >= 1 AND career_growth <= 5),
  company_culture   INT CHECK (company_culture >= 1 AND company_culture <= 5),
  title             VARCHAR(200) NOT NULL,
  description       TEXT,
  is_anonymous      BOOLEAN DEFAULT false,
  status            VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending','approved','rejected')),
  created_at        TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMP NOT NULL DEFAULT NOW(),
  UNIQUE(job_id, user_id)
);

Table: recr_candidate_reviews - Candidate Reviews

Description: Interviewer reviews and ratings for candidates after interviews.

CREATE TABLE recr_candidate_reviews (
  id                  BIGSERIAL PRIMARY KEY,
  candidate_id        BIGINT NOT NULL REFERENCES recr_candidates(id),
  interviewer_id      BIGINT NOT NULL REFERENCES recr_users(id),
  interview_id        BIGINT NOT NULL REFERENCES recr_interviews(id),
  technical_skills    INT CHECK (technical_skills >= 1 AND technical_skills <= 5),
  communication_skills INT CHECK (communication_skills >= 1 AND communication_skills <= 5),
  cultural_fit        INT CHECK (cultural_fit >= 1 AND cultural_fit <= 5),
  problem_solving     INT CHECK (problem_solving >= 1 AND problem_solving <= 5),
  team_collaboration  INT CHECK (team_collaboration >= 1 AND team_collaboration <= 5),
  title               VARCHAR(200) NOT NULL,
  description         TEXT,
  recommendation      VARCHAR(20) CHECK (recommendation IN ('strong_yes','yes','neutral','no','strong_no')),
  created_at          TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at          TIMESTAMP NOT NULL DEFAULT NOW(),
  UNIQUE(interview_id, interviewer_id)
);
